"""Compare regex-instruction conditions with the c1B baseline (all 119 cases).

Usage: regex_analyze.py LABEL [LABEL ...]   (reads regex-LABEL.json)
Problem type comes from the expected regex: \\d+ -> numbers, [aeiouAEIOU] -> vowels,
other regex -> literal; non-regex calls -> other. Estimates use the t6 CPU fit
t = 0.0791*F + 0.001984*L seconds (an estimate, not a CPU measurement).
"""
import json
import re
import sys
from pathlib import Path

S = Path(__file__).parent
BASE = Path('<repo>/docs/measurements/'
            '2026-11-18-instruction-shortening/gpu-accuracy-c1B-all.json')
runs = {'c1B': json.loads(BASE.read_text())}
runs |= {c: json.loads((S / f'regex-{c}.json').read_text()) for c in sys.argv[1:]}
TIER3 = {'u03-regex-digits', 'u04-regex-whitespace', 'u05-leading-space', 'u07-unicode'}


def tier(suite, case_id):
    if suite in ('changed5',) or (suite == 'unseen12' and case_id not in TIER3):
        return 2
    return 3 if suite in ('boundary3', 'unseen12') else 1


def kind(r):
    regex = r['expected_parameters'].get('regex')
    if regex is None:
        return 'other'
    return {'\\d+': 'numbers', '[aeiouAEIOU]': 'vowels'}.get(regex, 'literal')


def params(r):
    return json.loads(r['generated_text'])['parameters'] if r.get('completed') else None


def semantic(r):
    """Regex call: same name and source, and re.sub gives the expected text on
    the source string and on every fixture regex_check input."""
    p, e = params(r), r['expected_parameters']
    if p is None or r.get('selected_name') != r['expected_name']:
        return False
    if p.get('source_string') != e['source_string']:
        return False
    texts = [(e['source_string'], re.sub(e['regex'], e['replacement'], e['source_string']))]
    texts += [(c['input'], c['output']) for c in r.get('regex_checks', [])]
    try:
        return all(re.sub(p['regex'], p['replacement'], t) == out for t, out in texts)
    except re.error:
        return False


def estimate(run):
    s = run['summary']['standard11']
    f = s['forwards_select_name'] + s['forwards_generate_parameters']
    n = s['forward_length_sum_select_name'] + s['forward_length_sum_generate_parameters']
    return 0.0791 * f + 0.001984 * n, s['param_prompt_tokens']


print('label | tier1/99 tier2/13 tier3/7 | numbers vowels literal other (tier 1) | caps | std11 tokens est')
for c, run in runs.items():
    rows = [(k, r) for k, v in run['suites'].items() for r in v]
    t = {i: sum(bool(r['joint_correct']) for k, r in rows if tier(k, r['id']) == i) for i in (1, 2, 3)}
    kinds = {}
    for k, r in rows:
        if tier(k, r['id']) == 1:
            ok, n = kinds.get(kind(r), (0, 0))
            kinds[kind(r)] = (ok + bool(r['joint_correct']), n + 1)
    caps = sum('token limit' in (r.get('error') or '') for _, r in rows)
    sec, tok = estimate(run)
    sem = {x: sum(semantic(r) for k, r in rows if tier(k, r['id']) == 1 and kind(r) == x)
           for x in ('numbers', 'vowels', 'literal')}
    print(f'{c} | {t[1]} {t[2]} {t[3]} | ' + ' '.join(
        f'{kinds[x][0]}/{kinds[x][1]}' for x in ('numbers', 'vowels', 'literal', 'other'))
        + f' | {caps} | {tok} {sec:.1f}s | semantic numbers {sem["numbers"]}'
        f' vowels {sem["vowels"]} literal {sem["literal"]}')
print('--- per-case changes from c1B (joint change or different parameters)')
base = {(k, r['id']): r for k, v in runs['c1B']['suites'].items() for r in v}
for c, run in list(runs.items())[1:]:
    for k, v in run['suites'].items():
        for r in v:
            b = base[(k, r['id'])]
            if b['joint_correct'] != r['joint_correct'] or params(b) != params(r):
                mark = {(True, False): 'LOST', (False, True): 'GAINED'}.get(
                    (bool(b['joint_correct']), bool(r['joint_correct'])), 'both-wrong')
                print(f'{c} {r["id"]} [{mark}]')
                print('   c1B:', json.dumps(params(b), ensure_ascii=False))
                print(f'   {c}:', json.dumps(params(r), ensure_ascii=False), r.get('error') or '')
