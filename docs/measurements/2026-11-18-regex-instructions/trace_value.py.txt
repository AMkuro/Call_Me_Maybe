"""Show the value text the product writes for chosen cases (accuracy probe, not timing).

Usage: trace_value.py FIXTURE CASE_ID [CASE_ID ...]
Records every token chosen inside generate_value, so a run that hits the
256-token cap still shows what was written.
"""
import json
import sys
from pathlib import Path
from unittest.mock import patch

from src import pipeline
from src.errors import PromptError
from src.loader import index_functions
from src.model_port import EOS, Model
from src.name_trie import build_name_trie
from src.schema import FunctionDef
from src.vocab import load_vocab

data = json.loads(Path(sys.argv[1]).read_text())
model = Model()
vocab = load_vocab(Path(model.vocab_path()))
functions = index_functions(FunctionDef.model_validate(f) for f in data['functions'])
trie = build_name_trie(list(functions), model.encode, EOS)
resources = pipeline.GenerationResources(model=model, vocab=vocab, functions=functions, trie=trie)
real_value = pipeline.generate_value
real_select = pipeline.select_next_token
for case in data['cases']:
    if case['id'] not in sys.argv[2:]:
        continue
    values = []   # (parameter name, chosen token ids)
    inside = [False]

    def value(slot, ids, **kw):
        values.append((slot.name, []))
        inside[0] = True
        try:
            return real_value(slot, ids, **kw)
        finally:
            inside[0] = False

    def select(model_, ids, allowed, **kw):
        if not inside[0]:
            return real_select(model_, ids, allowed, **kw)
        # Diagnostic only: the raw argmax never changes the chosen token.
        result = real_select(model_, ids, allowed, include_raw_top=True)
        values[-1][1].append((result[0], result[2]))
        return result

    try:
        with patch.object(pipeline, 'generate_value', side_effect=value), \
                patch.object(pipeline, 'select_next_token', side_effect=select):
            call = pipeline.process_prompt(case['prompt'], resources)
        status = f'ok saved={call._generated_json}'
    except PromptError as e:
        status = f'error: {e}'
    print(case['id'], case['prompt'], status)
    for name, pairs in values:
        print(f'  {name} ({len(pairs)} tokens): {b"".join(vocab[c] for c, _ in pairs)[:60]!r}')
        for chosen_id, raw in pairs[:6]:
            want = vocab[raw] if raw is not None else '(forced)'
            print(f'    chose {vocab[chosen_id]!r} model wanted {want!r}')
