#!/usr/bin/env bash
# GPU accuracy runs on all 119 cases after the integer-delimiter grammar change
# (worktree code), in parallel. Accuracy only, not timing.
set -u
S=<scratch>
W=<worktree>
R=<repo>
P1='Convert descriptions of patterns into actual regular expressions, such as \d+ for a run of digits and [xyzXYZ] for any one of x, y or z in either case; do not list the matched substrings.'
P2='Convert descriptions of patterns into actual regular expressions, such as \d+ for numbers and [aeiouAEIOU] for vowels; do not list the matched substrings.'
cd "$W"
run() {
  L=$1; shift
  HF_HOME=$R/.cache/huggingface HF_HUB_OFFLINE=1 TRANSFORMERS_OFFLINE=1 PYTHONPATH=$W \
    $R/.venv-gpu/bin/python "$S/regex_exp.py" --label "$L" "$@" \
    --output "$S/regex-$L.json" > "$S/regex-$L.log" 2>&1
}
run n1 &
run p1 --replace "4=$P1" &
run p2 --replace "4=$P2" &
wait
for L in n1 p1 p2; do echo "$L: $(tail -n 1 "$S/regex-$L.log")"; done
