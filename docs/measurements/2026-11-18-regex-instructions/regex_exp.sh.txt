#!/usr/bin/env bash
# GPU accuracy runs of regex_exp.py on all 119 cases, in parallel (accuracy only, not timing).
# Code and fixtures come from the exp/regex-evaluation worktree; the GPU venv and
# model cache come from the main checkout (the worktree has neither).
set -u
S=<scratch>
W=<worktree>
R=<repo>
S6='Convert descriptions of patterns into actual regular expressions; do not list the matched substrings.'
RULE='Write a choice among single characters as a bracket class such as [abc], and one or more repeats with +.'
CASE='Letter classes cover both upper and lower case unless the request says otherwise.'
cd "$W"
run() {
  L=$1; shift
  HF_HOME=$R/.cache/huggingface HF_HUB_OFFLINE=1 TRANSFORMERS_OFFLINE=1 PYTHONPATH=$W \
    $R/.venv-gpu/bin/python "$S/regex_exp.py" --label "$L" "$@" \
    --output "$S/regex-$L.json" > "$S/regex-$L.log" 2>&1
}
run g1 --replace "4=$S6 $RULE" &
run g2 --replace "4=$S6 $RULE" --add "$CASE" &
run g3 --add "$CASE" &
wait
for L in g1 g2 g3; do echo "$L: $(tail -n 1 "$S/regex-$L.log")"; done
