"""Count number arguments whose expected value is written in the request (analysis only).

Usage: number_in_prompt.py FIXTURE [FIXTURE ...]
A value counts as written when some -?digits(.digits)? literal in the prompt
has the same Decimal value. Prints every case where it is not.
"""
import json
import re
import sys
from decimal import Decimal
from pathlib import Path

LITERAL = re.compile(r'-?\d+(?:\.\d+)?')
found = total = 0
for path in sys.argv[1:]:
    data = json.loads(Path(path).read_text())
    types = {f['name']: {k: v['type'] for k, v in f['parameters'].items()}
             for f in data['functions']}
    for case in data['cases']:
        params = case.get('expected_parameters')
        if params is None:
            continue
        written = {Decimal(m) for m in LITERAL.findall(case['prompt'])}
        for name, value in params.items():
            if types[case['expected_name']].get(name) != 'number':
                continue
            total += 1
            if Decimal(str(value)) in written:
                found += 1
            else:
                print(f'not written: {case["id"]} {name}={value!r} prompt={case["prompt"]!r}')
print(f'{found}/{total} number arguments are written in the request')
