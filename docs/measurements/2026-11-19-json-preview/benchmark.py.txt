import io, contextlib, time, json, statistics
from unittest.mock import patch
from src.display import Dashboard
from src.model_port import Model

def trial(interactive, frames, records=100):
    stream=io.StringIO()
    with contextlib.redirect_stderr(stream):
        dashboard=Dashboard(1,object.__new__(Model),{})
    dashboard.interactive=interactive
    clock=[1000.]
    with patch('src.display.time.monotonic',side_effect=lambda: clock[0]):
        start=time.perf_counter()
        for record in range(records):
            dashboard.stage('generate_call_json')
            for index in range(500):
                clock[0] += frames
                dashboard.output_token(b'x',token=1,mode='forced')
            stream.seek(0); stream.truncate()
        return time.perf_counter()-start
rows={}
for name,interactive,frames in [('redirected',False,0),('tty_throttled',True,0),('tty_500_tokens_per_second',True,.002),('tty_every_token',True,.1)]:
    records=10 if name == 'tty_every_token' else 100
    values=[trial(interactive,frames,records) for _ in range(3)]
    rows[name]={'tokens_per_trial':records*500,'seconds':values,'median_us_per_token':statistics.median(values)*1e6/(records*500)}
print(json.dumps({'notes':'Synthetic observer-only cost; 100 prefixes of 500 ASCII tokens; three serial trials, no outliers removed; no model or hardware probe; no +8s offset. Render workload simulates 500 tokens/s and 20fps, substantially faster than measured real GPU token rate. StringIO excludes terminal emulator cost.', 'results':rows},indent=2))
