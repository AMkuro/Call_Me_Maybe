"""Print the parameter prompt of each condition and token counts (tokenizer only)."""
import json
import sys
from pathlib import Path

sys.argv = ['x', '--condition', 'A', '--output', '/dev/null']
from src.loader import index_functions  # noqa: E402
from src.model_port import Model  # noqa: E402
from src.prompt import build_name_prompt, PARAMETER_INSTRUCTIONS  # noqa: E402
from src.schema import FunctionDef  # noqa: E402

exp = {}
exec(Path(__file__).with_name('prompt_exp.py').read_text().split('parser = ')[0], exp)
model = Model()
data = json.loads(Path('docs/measurements/2026-11-18-req031/standard.json').read_text())
functions = index_functions(FunctionDef.model_validate(f) for f in data['functions'])
case = data['cases'][0]
fn = functions[case['expected_name']]
for c in 'ABCD':
    text = exp['param_prompt'](case['prompt'], fn, c)
    print(f'--- {c} ({len(model.encode(text))} tokens)\n{text}')
s = exp['short_prompt'](fn)
print(f'--- DYN after the prompt field ({len(model.encode(s))} tokens)\n{s}')
print('--- parts (tokens)')
print('instructions', len(model.encode(PARAMETER_INSTRUCTIONS)))
print('request line', len(model.encode(f'User request: {json.dumps(case["prompt"])}\n')))
for c in 'ABCD':
    print('definition', c, len(model.encode(exp['definition'](fn, exp['KEEP'][c]))))
print('name prompt', len(model.encode(build_name_prompt(case['prompt'], functions))))
tot = {c: 0 for c in [*'ABCD', 'S']}
for case in data['cases']:
    fn = functions[case['expected_name']]
    for c in 'ABCD':
        tot[c] += len(model.encode(exp['param_prompt'](case['prompt'], fn, c)))
    tot['S'] += len(model.encode(exp['short_prompt'](fn)))
print('standard11 parameter-prompt token totals', tot)
