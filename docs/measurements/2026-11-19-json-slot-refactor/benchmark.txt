"""Run from the repository root with PYTHONPATH=. .venv/bin/python <this file>."""
import ast
import json
import statistics
import subprocess
import sys
import timeit
import types
from pathlib import Path

from src import values
from src.vocab import load_vocab

baseline = '0400edbc07afd54c8a8ca6f8d3e45a75d389de45'
source = subprocess.check_output(
    ['git', 'show', baseline + ':src/values.py'], text=True,
)
before = types.ModuleType('src.values_before')
exec(compile(source, 'values_before.py', 'exec'), before.__dict__)
# The working tree initially differed from HEAD only in formatting.
snapshot = Path('/tmp/cmm-values-before.py')
if snapshot.exists():
    assert ast.dump(ast.parse(source)) == ast.dump(
        ast.parse(snapshot.read_text()),
    )
vocab_path = next(Path('/home/miwasaki/.cache/huggingface/hub').glob(
    'models--Qwen--Qwen3-0.6B/snapshots/c1899de*/vocab.json',
))
vocab = load_vocab(vocab_path)
rows = []
for size in (0, 128, 8192):
    for pending in (b'', b'\xe3\x81'):
        written = b'"' + b'a' * size + pending
        assert before.allowed_slot_tokens(vocab, 'string', written, '}') == (
            values.allowed_slot_tokens(vocab, 'string', written, '}')
        )
        times = {'before': [], 'after': []}
        for repeat in range(9):
            pairs = [('before', before), ('after', values)]
            if repeat % 2:
                pairs.reverse()
            for name, module in pairs:
                elapsed = timeit.timeit(
                    lambda: module.allowed_slot_tokens(
                        vocab, 'string', written, '}',
                    ), number=3,
                ) / 3
                times[name].append(elapsed)
        rows.append({
            'written_bytes': len(written), 'pending_utf8': pending.hex(),
            'seconds': times,
            'median_ms': {name: statistics.median(samples) * 1000
                          for name, samples in times.items()},
        })
print(json.dumps({
    'python': sys.version, 'baseline': baseline,
    'vocab_path': str(vocab_path), 'vocab_size': len(vocab),
    'rows': rows,
}, indent=2))
