import json
from pathlib import Path
from src.model_port import Model
from src.vocab import load_vocab
m=Model()
v=load_vocab(Path(m.vocab_path()))
rows=[]
for text in ['"𠮷🦄🧑🏽‍💻"}', '"前<|im_end|>後"', '"前\\u003c|im_end|>後"']:
 ids=m.encode(text)
 partial=[]
 for t in ids:
  if t in v:
   try: v[t].decode('utf8')
   except UnicodeDecodeError: partial.append({'id':t,'bytes':list(v[t])})
 rows.append({'text':text,'ids':ids,'decoded':m.decode(ids),'partial':partial})
print(json.dumps(rows,ensure_ascii=False,indent=2))
assert rows[0]['partial'] and rows[0]['decoded']==rows[0]['text']
assert json.loads(rows[1]['text'])==json.loads(rows[2]['text'])
