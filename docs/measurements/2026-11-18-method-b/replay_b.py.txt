"""Method B replay: one forward before every output token (measurement only).

Phase 1 runs the current pipeline unchanged and records, through wrappers
around our own modules and the public SDK calls, every logits call (input
length, seconds), every mask computation and the exact output IDs.
Phase 2 replays those IDs with teacher forcing: for each output position it
calls the public logits API once with prompt + output[:i], as method B
would, and times it. Phase 3 profiles a few forwards with the standard
library cProfile (call-stack observation only; no SDK attribute access).
No torch/transformers import here; the SDK is used only via src.model_port.
Run with cwd = repository root.
"""

import argparse
import cProfile
import hashlib
import io
import json
import os
import platform
import pstats
import subprocess
import sys
import time
from pathlib import Path

import numpy as np

import src.decoder as decoder
import src.pipeline as pipeline
from src.loader import load_functions, load_prompts
from src.model_port import EOS, Model
from src.name_trie import build_name_trie
from src.pipeline import GenerationResources, process_prompt
from src.prompt import build_name_prompt, build_parameters_prompt
from src.vocab import load_vocab

EVENTS: list[tuple] = []
CHOOSE = decoder.choose_token  # unwrapped original


class RecordingModel(Model):
    """Public-API wrapper that logs calls in order."""

    def logits(self, ids: list[int]) -> list[float]:
        t = time.perf_counter()
        out = super().logits(ids)
        EVENTS.append(("logits", len(ids), time.perf_counter() - t))
        return out

    def decode(self, ids: list[int]) -> str:
        t = time.perf_counter()
        out = super().decode(ids)
        EVENTS.append(("decode", list(ids), time.perf_counter() - t))
        return out

    def encode(self, text: str) -> list[int]:
        t = time.perf_counter()
        out = super().encode(text)
        EVENTS.append(("encode", len(out), time.perf_counter() - t))
        return out


def _wrap(module, name, tag, size=None):
    original = getattr(module, name)

    def wrapper(*args, **kwargs):
        t = time.perf_counter()
        out = original(*args, **kwargs)
        EVENTS.append((tag, size(out) if size else None,
                       time.perf_counter() - t))
        return out
    setattr(module, name, wrapper)


def sha256(path: Path) -> str:
    return hashlib.sha256(path.read_bytes()).hexdigest()


def stats(values: list[float]) -> dict:
    if not values:
        return {"n": 0}
    a = np.asarray(values)
    return {"n": len(values), "sum": float(a.sum()), "mean": float(a.mean()),
            "median": float(np.median(a)), "min": float(a.min()),
            "max": float(a.max())}


def secs(events: list[tuple], tag: str) -> list[float]:
    return [e[2] for e in events if e[0] == tag]


def main() -> int:
    parser = argparse.ArgumentParser()
    parser.add_argument("--output", type=Path, required=True)
    parser.add_argument("--profile-forwards", type=int, default=5)
    parser.add_argument("--first", type=int, default=1, help="1-based")
    parser.add_argument("--last", type=int, help="1-based, inclusive")
    args = parser.parse_args()
    report: dict = {
        "measurement": "method-B teacher-forcing replay; CPU timing",
        "started_utc": time.strftime("%Y-%m-%dT%H:%M:%SZ", time.gmtime()),
        "head": subprocess.run(["git", "rev-parse", "HEAD"], text=True,
                               capture_output=True).stdout.strip(),
        "dirty_src_sdk": subprocess.run(
            ["git", "status", "--short", "src", "llm_sdk"],
            text=True, capture_output=True).stdout,
        "python": platform.python_version(),
        "env": {k: os.environ.get(k) for k in (
            "OMP_NUM_THREADS", "CUDA_VISIBLE_DEVICES", "HF_HUB_OFFLINE")},
        "sched_affinity": sorted(os.sched_getaffinity(0)),
        "source_sha256": {str(p): sha256(p) for p in sorted(
            list(Path("src").glob("*.py"))
            + [Path("llm_sdk/llm_sdk/__init__.py")])},
        "input_sha256": {p: sha256(Path(p)) for p in (
            "data/input/functions_definition.json",
            "data/input/function_calling_tests.json")},
    }

    t0 = time.perf_counter()
    functions = load_functions(Path("data/input/functions_definition.json"))
    prompts = load_prompts(Path("data/input/function_calling_tests.json"))
    selected = list(enumerate(prompts, 1))[args.first - 1:args.last]
    report["selected_indices"] = [i for i, _ in selected]
    t1 = time.perf_counter()
    model = RecordingModel()
    t2 = time.perf_counter()
    vocab = load_vocab(Path(model.vocab_path()))
    trie = build_name_trie(list(functions), model.encode, EOS)
    resources = GenerationResources(model=model, vocab=vocab,
                                    functions=functions, trie=trie)
    t3 = time.perf_counter()
    report["startup_s"] = {"inputs": t1 - t0, "model_load": t2 - t1,
                           "vocab_trie_resources": t3 - t2}

    _wrap(pipeline, "allowed_value_tokens", "allowed_value", len)
    _wrap(pipeline, "next_tokens", "allowed_name", len)
    _wrap(decoder, "choose_token", "choose")
    for stage in ("select_name", "generate_parameters", "validate_call"):
        _wrap(pipeline, stage, "stage:" + stage)

    # Phase 1: current implementation, unchanged.
    cases = []
    phase1_start = time.perf_counter()
    for index, prompt in selected:
        EVENTS.clear()
        t = time.perf_counter()
        call = process_prompt(prompt, resources)
        wall = time.perf_counter() - t
        events = list(EVENTS)
        ends = [i for i, e in enumerate(events) if e[0].startswith("stage:")]
        name_ev, param_ev = events[:ends[0]], events[ends[0] + 1:ends[1]]
        fn = functions[call.name]
        name_prompt = model.encode(build_name_prompt(prompt, functions))
        param_prompt = model.encode(build_parameters_prompt(prompt, fn))
        path = list(trie.paths[call.name])
        output_ids = [e for e in param_ev if e[0] == "decode"][-1][1]
        name_allowed = [e[1] for e in name_ev if e[0] == "allowed_name"]
        value_allowed = [e[1] for e in param_ev if e[0] == "allowed_value"]
        cases.append({
            "index": index, "prompt": prompt, "function": call.name,
            "generated_text": call._generated_json,
            "wall_s": wall,
            "stage_s": {e[0][6:]: e[2] for e in events
                        if e[0].startswith("stage:")},
            "name": {
                "prompt_tokens": len(name_prompt), "path": path,
                "output_tokens": len(path),
                "single_candidate": sum(1 for n in name_allowed if n == 1),
                "branch": sum(1 for n in name_allowed if n > 1),
                "forward_lengths_current": [e[1] for e in name_ev
                                            if e[0] == "logits"],
                "logits_s": stats(secs(name_ev, "logits")),
                "mask_s": stats(secs(name_ev, "allowed_name")),
                "choose_s": stats(secs(name_ev, "choose")),
                "encode_s": stats(secs(name_ev, "encode")),
            },
            "parameters": {
                "prompt_tokens": len(param_prompt), "output_ids": output_ids,
                "output_tokens": len(output_ids),
                "value_tokens": len(value_allowed),
                "value_single_candidate": sum(1 for n in value_allowed
                                              if n == 1),
                "value_branch": sum(1 for n in value_allowed if n > 1),
                "value_allowed_sizes": value_allowed,
                "fixed_json_tokens": len(output_ids) - len(value_allowed),
                "forward_lengths_current": [e[1] for e in param_ev
                                            if e[0] == "logits"],
                "logits_s": stats(secs(param_ev, "logits")),
                "mask_s": stats(secs(param_ev, "allowed_value")),
                "choose_s": stats(secs(param_ev, "choose")),
                "encode_s": stats(secs(param_ev, "encode")),
                "decode_s": stats(secs(param_ev, "decode")),
            },
        })
        print(f"phase1 {index}/{len(prompts)} {wall:.2f}s", file=sys.stderr,
              flush=True)
    report["cases"] = cases
    report["phase1_current_total_s"] = time.perf_counter() - phase1_start

    # Phase 2: method B replay with the recorded IDs (teacher forcing).
    phase2_start = time.perf_counter()
    for case in cases:
        fn = functions[case["function"]]
        name_prompt = model.encode(build_name_prompt(case["prompt"],
                                                     functions))
        param_prompt = model.encode(build_parameters_prompt(case["prompt"],
                                                            fn))
        for key, prompt_ids, out in (
                ("name", name_prompt, case["name"]["path"]),
                ("parameters", param_prompt,
                 case["parameters"]["output_ids"])):
            lengths, forward_s, choose_s = [], [], []
            for i, token in enumerate(out):
                ids = prompt_ids + out[:i]
                t = time.perf_counter()
                logits = Model.logits(model, ids)
                forward_s.append(time.perf_counter() - t)
                t = time.perf_counter()
                if CHOOSE(logits, [token]) != token:
                    raise SystemExit("replay mask failed")
                choose_s.append(time.perf_counter() - t)
                lengths.append(len(ids))
            case[key]["method_b"] = {
                "forwards": len(out), "lengths": lengths,
                "max_length": max(lengths), "forward_s": forward_s,
                "forward_total_s": sum(forward_s),
                "choose_single_s": stats(choose_s),
            }
        print(f"phase2 {case['index']} "
              f"name={case['name']['method_b']['forward_total_s']:.1f}s "
              f"params={case['parameters']['method_b']['forward_total_s']:.1f}s",
              file=sys.stderr, flush=True)
        # Save after each case so a safety stop keeps finished cases.
        args.output.write_text(json.dumps(report, ensure_ascii=False,
                                          indent=1))
    report["phase2_replay_total_s"] = time.perf_counter() - phase2_start

    # Phase 3: cProfile of a few forwards at the longest method-B length.
    longest = max(cases,
                  key=lambda c: c["parameters"]["method_b"]["max_length"])
    fn = functions[longest["function"]]
    ids = (model.encode(build_parameters_prompt(longest["prompt"], fn))
           + longest["parameters"]["output_ids"][:-1])
    profiler = cProfile.Profile()
    t = time.perf_counter()
    for _ in range(args.profile_forwards):
        profiler.runcall(Model.logits, model, ids)
    report["profile"] = {"length": len(ids),
                         "forwards": args.profile_forwards,
                         "wall_s": time.perf_counter() - t}
    buf = io.StringIO()
    ps = pstats.Stats(profiler, stream=buf)
    ps.sort_stats("cumulative").print_stats(45)
    ps.sort_stats("tottime").print_stats(30)
    report["profile"]["text"] = buf.getvalue()

    # Phase 4: pure-Python costs on a vocabulary-sized logits list.
    rng = np.random.default_rng(0)
    fake = [float(x) for x in rng.standard_normal(151936)]

    def bench(fn_, n=20):
        out = []
        for _ in range(n):
            t = time.perf_counter()
            fn_()
            out.append(time.perf_counter() - t)
        return stats(out)
    report["python_microbench_s"] = {
        "float_listcomp_151936": bench(lambda: [float(x) for x in fake]),
        "np_asarray_151936": bench(lambda: np.asarray(fake, dtype=float)),
        "choose_token_1_allowed": bench(lambda: CHOOSE(fake, [5])),
        "choose_token_100k_allowed": bench(
            lambda: CHOOSE(fake, range(100000))),
    }
    report["finished_utc"] = time.strftime("%Y-%m-%dT%H:%M:%SZ",
                                           time.gmtime())
    args.output.write_text(json.dumps(report, ensure_ascii=False, indent=1))
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
