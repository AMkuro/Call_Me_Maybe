#!/usr/bin/env bash
# GPU accuracy runs of instr_exp.py; five at a time (accuracy only, not timing).
# Usage: instr_exp.sh LABEL:DROP:DEF[:I=TEXT] ...   e.g. d3:3:A   r2::A:2=Copy ...
# TEXT must not contain ':'.
set -u
S=<scratch>
cd <repo>
run() {
  IFS=: read -r L DROP DEF REPL <<< "$1"
  extra=()
  if [ -n "$REPL" ]; then extra=(--replace "$REPL"); fi
  # SUITES (environment, optional): comma-separated suite names for instr_exp.py.
  if [ -n "${SUITES:-}" ]; then extra+=(--suites "$SUITES"); fi
  HF_HOME=<repo>/.cache/huggingface HF_HUB_OFFLINE=1 \
    TRANSFORMERS_OFFLINE=1 PYTHONPATH=<repo> \
    <repo>/.venv-gpu/bin/python "$S/instr_exp.py" \
    --label "$L" --drop "$DROP" --definition "$DEF" "${extra[@]}" \
    --output "$S/instr-$L.json" > "$S/instr-$L.log" 2>&1
}
n=0
for spec in "$@"; do
  run "$spec" &
  n=$((n + 1))
  if [ "$n" -ge 5 ]; then wait; n=0; fi
done
wait
for spec in "$@"; do L=${spec%%:*}; echo "$L: $(tail -n 1 "$S/instr-$L.log")"; done
