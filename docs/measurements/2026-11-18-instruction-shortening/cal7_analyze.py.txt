"""Seven calibrated suites: exact match, regex_checks semantics, diffs from A.

Usage: cal7_analyze.py LABEL [LABEL ...]   (reads instr-LABEL-cal7.json; A is the reference)
Semantic check is separate from exact match: a regex call passes when
re.sub(regex, replacement, input) == output for every fixture regex_check.
"""
import json
import re
import sys
from pathlib import Path

S = Path(__file__).parent
labels = sys.argv[1:]
runs = {c: json.loads((S / f'instr-{c}-cal7.json').read_text()) for c in ['A', *labels]}


def params(r):
    return json.loads(r['generated_text'])['parameters'] if r.get('completed') else None


def semantic(r):
    """None if not a regex case; else whether every regex_check passes."""
    if 'regex_checks' not in r:
        return None
    p = params(r)
    if p is None or r.get('selected_name') != r['expected_name']:
        return False
    try:
        return all(re.sub(p['regex'], p['replacement'], c['input']) == c['output']
                   for c in r['regex_checks']) and p['source_string'] == r['expected_parameters']['source_string']
    except (re.error, KeyError):
        return False


print('label | exact per suite | exact /77 | regex exact | regex semantic | caps | errors | prompt tokens')
for c, run in runs.items():
    rows = [r for v in run['suites'].values() for r in v]
    reg = [r for r in rows if 'regex_checks' in r]
    print(f'{c} | {" ".join(str(s["joint_correct"]) for s in run["summary"].values())} | '
          f'{sum(bool(r["joint_correct"]) for r in rows)} | '
          f'{sum(bool(r["joint_correct"]) for r in reg)}/{len(reg)} | '
          f'{sum(bool(semantic(r)) for r in reg)}/{len(reg)} | '
          f'{sum("token limit" in r.get("error", "") for r in rows)} | '
          f'{sum(not r.get("completed") for r in rows)} | '
          f'{sum(s["param_prompt_tokens"] for s in run["summary"].values())}')
print('--- A wrong cases')
for k, rows in runs['A']['suites'].items():
    for r in rows:
        if not r['joint_correct']:
            print(f'A {r["id"]}: {json.dumps(params(r), ensure_ascii=False)} expected '
                  f'{json.dumps(r["expected_parameters"], ensure_ascii=False)} semantic={semantic(r)}')
print('--- per-case differences from A')
for c in labels:
    for k, rows in runs['A']['suites'].items():
        for a, n in zip(rows, runs[c]['suites'][k]):
            if a.get('joint_correct') != n.get('joint_correct') or params(a) != params(n):
                mark = {(True, False): 'LOST', (False, True): 'GAINED'}.get(
                    (bool(a['joint_correct']), bool(n['joint_correct'])), 'both-wrong')
                print(f'{c} {a["id"]} [{mark}] semantic {semantic(a)}->{semantic(n)}')
                print('   A :', json.dumps(params(a), ensure_ascii=False))
                print(f'   {c}:', json.dumps(params(n), ensure_ascii=False), n.get('error', ''))
